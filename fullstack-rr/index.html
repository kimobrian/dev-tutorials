app.ts — the middleware stack

src/app.tsts
import express from 'express'
import cors    from 'cors'
import helmet  from 'helmet'
import morgan  from 'morgan'
import rateLimit from 'express-rate-limit'
import { errorHandler } from './middleware/errorHandler'
import authRoutes  from './routes/auth.routes'
import postsRoutes from './routes/posts.routes'

const app = express()

// Security headers — prevents XSS, clickjacking, MIME sniffing
app.use(helmet())

// CORS — allow our two frontend origins
app.use(cors({
  origin: ['http://localhost:5173', 'http://localhost:5174', process.env.FRONTEND_URL ?? ''],
  credentials: true,
}))

// Rate limiting — 100 req / 15 min per IP
app.use(rateLimit({ windowMs: 15 * 60 * 1000, max: 100 }))

app.use(morgan('dev'))                // request logging
app.use(express.json())              // parse JSON bodies
app.use(express.urlencoded({ extended: true }))

// Routes
app.get('/health', (_, res) => res.json({ status: 'ok', ts: new Date().toISOString() }))
app.use('/api/auth',  authRoutes)
app.use('/api/posts', postsRoutes)

// 404 handler — must be after all routes
app.use((_, res) => res.status(404).json({ error: 'Not found' }))

// Global error handler — must have 4 params for Express to recognise it
app.use(errorHandler)

export default app

import { Request, Response, NextFunction, ErrorRequestHandler } from 'express'

export class AppError extends Error {
  constructor(public statusCode: number, message: string) {
    super(message); this.name = 'AppError'
  }
}

export const errorHandler: ErrorRequestHandler = (err, _req, res, _next) => {
  if (err instanceof AppError) {
    return res.status(err.statusCode).json({ error: err.message })
  }
  console.error(err)
  res.status(500).json({ error: 'Internal server error' })
}

Authentication — JWT + bcrypt

  
Stateless auth with a short-lived access token (15 min) and a long-lived refresh token (7 days). The refresh token is stored in an httpOnly cookie — inaccessible to JavaScript, safe from XSS.

  
jsonwebtokenbcrypthttpOnly cookieRefresh rotationAuth middleware


  
Token Strategy
Access token (15 min) — sent in Authorization header. Expires fast to limit damage if stolen. Refresh token (7 days) — stored in httpOnly cookie, never readable by JS. When the access token expires, POST /auth/refresh exchanges the cookie for a new access token.


  
Auth service — the business logic

  
src/services/auth.service.tsts
import bcrypt from 'bcrypt'
import jwt    from 'jsonwebtoken'
import { User } from '../models'
import { env }  from '../config/env'
import { AppError } from '../middleware/errorHandler'

const SALT_ROUNDS = 12

export async function register(email: string, name: string, password: string) {
  const exists = await User.findOne({ where: { email } })
  if (exists) throw new AppError(409, 'Email already in use')

  const hash = await bcrypt.hash(password, SALT_ROUNDS)
  const user = await User.create({ email, name, password: hash })
  return issueTokens(user)
}

export async function login(email: string, password: string) {
  const user = await User.findOne({ where: { email } })
  if (!user) throw new AppError(401, 'Invalid credentials')
  const match = await bcrypt.compare(password, user.password)
  if (!match) throw new AppError(401, 'Invalid credentials')  // same message — no user enum
  return issueTokens(user)
}

function issueTokens(user: User) {
  const payload = { id: user.id, email: user.email, role: user.role }
  const accessToken  = jwt.sign(payload, env.jwtSecret,  { expiresIn: '15m' })
  const refreshToken = jwt.sign({ id: user.id }, env.jwtRefresh, { expiresIn: '7d' })
  return { accessToken, refreshToken, user: { id: user.id, name: user.name, role: user.role } }
}

import { Request, Response, NextFunction } from 'express'
import * as authService from '../services/auth.service'

const COOKIE_OPTS = {
  httpOnly: true,      // JS cannot read this cookie
  secure: process.env.NODE_ENV === 'production',
  sameSite: 'lax' as const,
  maxAge: 7 * 24 * 60 * 60 * 1000    // 7 days in ms
}

export async function register(req: Request, res: Response, next: NextFunction) {
  try {
    const { accessToken, refreshToken, user } = await authService.register(
      req.body.email, req.body.name, req.body.password
    )
    res.cookie('refreshToken', refreshToken, COOKIE_OPTS)
    res.status(201).json({ accessToken, user })
  } catch (e) { next(e) }
}

Auth middleware — protecting routes

  
src/middleware/auth.tsts
import { Request, Response, NextFunction } from 'express'
import jwt from 'jsonwebtoken'
import { env } from '../config/env'

// Extend Express Request to carry user
declare global { namespace Express { interface Request { user?: { id: number; role: string } } } }

export function authenticate(req: Request, res: Response, next: NextFunction) {
  const token = req.headers.authorization?.replace('Bearer ', '')
  if (!token) return res.status(401).json({ error: 'No token provided' })
  try {
    req.user = jwt.verify(token, env.jwtSecret) as { id: number; role: string }
    next()
  } catch {
    res.status(401).json({ error: 'Invalid or expired token' })
  }
}

export function requireAdmin(req: Request, res: Response, next: NextFunction) {
  if (req.user?.role !== 'admin') return res.status(403).json({ error: 'Admin only' })
  next()
}

Posts service — list with pagination & search

  
src/services/posts.service.tsts
import { Op } from 'sequelize'
import { Post, User, Comment, Tag } from '../models'

export async function listPosts({
  page = 1, limit = 10, q = '', tag = '', authorId
}: { page?: number; limit?: number; q?: string; tag?: string; authorId?: number }) {
  const where: any = { published: true }
  if (q) where[Op.or] = [
    { title: { [Op.iLike]: `%${q}%` } },   // case-insensitive LIKE
    { body:  { [Op.iLike]: `%${q}%` } },
  ]
  if (authorId) where.userId = authorId

  const { rows: posts, count } = await Post.findAndCountAll({
    where,
    include: [
      { model: User,    as: 'author',   attributes: ['id', 'name', 'avatarUrl'] },
      { model: Tag,     as: 'tags',     attributes: ['id', 'name'], // only if tag filter
        required: !!tag, where: tag ? { name: tag } : undefined },
    ],
    order:  [['createdAt', 'DESC']],
    limit:  limit,
    offset: (page - 1) * limit,
    distinct: true,   // required for correct count with hasMany includes
  })
  return { posts, total: count, pages: Math.ceil(count / limit), page }
}

export function uploadToCloudinary(buffer: Buffer, folder: string): Promise<string> {
  return new Promise((resolve, reject) => {
    const stream = cloudinary.uploader.upload_stream(
      { folder, transformation: [{ width: 1200, crop: 'limit', quality: 'auto' }] },
      (err, result) => { if (err || !result) reject(err); else resolve(result.secure_url) }
    )
    stream.end(buffer)
  })
}

Why integration over unit? Unit-testing controllers in isolation misses the thing most likely to break: the interaction between route parsing, middleware, business logic, and the ORM. Integration tests catch real bugs. Keep them fast with an in-memory SQLite test DB or a local Postgres instance.

  baseURL:     import.meta.env.VITE_API_URL ?? 'http://localhost:3001',

api.interceptors.request.use((config) => {
  const token = localStorage.getItem('accessToken')
  if (token) config.headers.Authorization = `Bearer ${token}`
  return config
})

let isRefreshing = false
let refreshQueue: Array<(token: string) => void> = []

// 401 → try refresh once, then replay failed requests
api.interceptors.response.use(
  (r) => r,
  async (error) => {
    const original = error.config
    if (error.response?.status !== 401 || original._retry) throw error
    original._retry = true
    if (isRefreshing) {
      return new Promise(res => refreshQueue.push(t => { original.headers.Authorization = `Bearer ${t}`; res(api(original)) }))
    }
    isRefreshing = true
    try {
      const { data } = await api.post('/api/auth/refresh')   // cookie sent automatically
      localStorage.setItem('accessToken', data.accessToken)
      refreshQueue.forEach(fn => fn(data.accessToken))
      refreshQueue = []
      return api(original)
    } catch { localStorage.removeItem('accessToken'); window.location.href = '/login' }
    finally  { isRefreshing = false }
  }
)

import { redirect } from 'react-router'

export function requireAuth({ role }: { role?: 'admin'|'user' } = {}) {
  const token = localStorage.getItem('accessToken')
  if (!token) throw redirect('/login')

  // Decode payload (public — not verification, just reading claims)
  const payload = JSON.parse(atob(token.split('.')[1]))
  if (role === 'admin' && payload.role !== 'admin') throw redirect('/')
  return payload
}

export function InfinitePostList({ initialPosts, totalPages }: Props) {
  const [posts, setPosts] = useState(initialPosts)
  const [page,  setPage]  = useState(1)
  const fetcher  = useFetcher()
  const sentinel = useRef<HTMLDivElement>(null)

  useEffect(() => {
    if (fetcher.data?.posts) {
      setPosts(prev => [...prev, ...fetcher.data.posts])  // append — never replace
    }
  }, [fetcher.data])

  useEffect(() => {
    const obs = new IntersectionObserver(([entry]) => {
      if (entry.isIntersecting && page < totalPages && fetcher.state === 'idle') {
        const next = page + 1; setPage(next)
        fetcher.load(`/?index&page=${next}`)   // calls the index loader
      }
    })
    if (sentinel.current) obs.observe(sentinel.current)
    return () => obs.disconnect()
  }, [page, totalPages, fetcher.state])

  return (
    <>
      {posts.map(p => <PostCard key={p.id} {...p} />)}
      <div ref={sentinel}>
        {fetcher.state !== 'idle' && <p>Loading more…</p>}
      </div>
    </>
  )
}

Loaders + Route Context

  
TanStack Router's loader runs before the component renders — just like RR v7. The key difference: the loader receives context including your QueryClient, so you can prefetch into the query cache rather than returning data directly.

The Pattern
The loader calls queryClient.ensureQueryData() — which returns cached data instantly if fresh, or fetches if stale. The component then calls useQuery() with the same key. They share one cache entry — no duplicate requests.

const postsQuery = (page: number, q: string) => ({
  queryKey: ['posts', { page, q }],
  queryFn:  () => api.get(`/api/posts?page=${page}&q=${q}`).then(r => r.data),

  validateSearch: searchSchema,

  // Runs before render — prefetches OR returns cached data
  loader: async ({ context: { queryClient }, deps }) => {
    await queryClient.ensureQueryData(postsQuery(deps.page, deps.q))

src/hooks/useLikePost.ts — optimistic like with rollbackts
export function useLikePost(postId: number) {
  const qc = useQueryClient()
  return useMutation({
    mutationFn: () => api.post(`/api/posts/${postId}/like`),

    onMutate: async () => {
      await qc.cancelQueries({ queryKey: keys.posts.detail(postId) })
      const prev = qc.getQueryData(keys.posts.detail(postId))

      // Optimistically update the cache BEFORE network request
      qc.setQueryData(keys.posts.detail(postId), (old: any) => ({
        ...old, likes: old.likes + 1, likedByMe: true
      }))
      return { prev }    // returned context goes to onError
    },

    onError: (err, _vars, ctx) => {
      // Roll back to the snapshot taken in onMutate
      qc.setQueryData(keys.posts.detail(postId), ctx?.prev)
    },

    onSettled: () => {
      // Always refetch to sync with server truth
      qc.invalidateQueries({ queryKey: keys.posts.detail(postId) })
    },
  })
}

useInfiniteQuery — Cursor Pagination

  
useInfiniteQuery accumulates pages in one cache entry. fetchNextPage appends the next chunk. An IntersectionObserver on a sentinel element triggers it automatically when the user scrolls to the bottom.

src/hooks/useInfinitePosts.tsts
import { useInfiniteQuery } from '@tanstack/react-query'

export function useInfinitePosts(q = '') {
  return useInfiniteQuery({
    queryKey: ['posts', 'infinite', { q }],
    queryFn:  ({ pageParam = 1 }) =>
      api.get(`/api/posts?page=${pageParam}&q=${q}&limit=10`).then(r => r.data),

    // Tell TanStack Query what the NEXT page param is
    // Return undefined when there are no more pages
    getNextPageParam: (lastPage) =>
      lastPage.page < lastPage.pages ? lastPage.page + 1 : undefined,

    initialPageParam: 1,
    staleTime: 30_000,
  })
}

// data.pages = array of page responses, each with .posts array
// Flatten: data.pages.flatMap(page => page.posts)


  
src/components/InfinitePostList.tsxtsx
export function InfinitePostList() {
  const { data, fetchNextPage, hasNextPage, isFetchingNextPage } = useInfinitePosts()
  const sentinel = useRef<HTMLDivElement>(null)

  useEffect(() => {
    const obs = new IntersectionObserver(([e]) => {
      if (e.isIntersecting && hasNextPage && !isFetchingNextPage) fetchNextPage()
    })
    if (sentinel.current) obs.observe(sentinel.current)
    return () => obs.disconnect()
  }, [hasNextPage, isFetchingNextPage, fetchNextPage])

  const posts = data?.pages.flatMap(p => p.posts) ?? []
  return (
    <>
      {posts.map(p => <PostCard key={p.id} {...p} />)}
      <div ref={sentinel}>{isFetchingNextPage && 'Loading…'}</div>
    </>
  )
}
